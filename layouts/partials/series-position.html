{{/* Where a page sits in its series ("series" in front matter). Returns a dict with name, parts (all posts of the series, oldest first), index (0-based), total and overview (the series page, if Hugo has one), or nothing for a page outside a series. Shared by the series label and the series navigation. */}} {{- $out := dict -}} {{- with .Params.series -}} {{- $name := . -}} {{- $parts := (where (where site.RegularPages "Section" "blog") "Params.series" $name).ByDate -}} {{- $index := -1 -}} {{- range $i, $p := $parts -}} {{- if eq $p.RelPermalink $.RelPermalink }}{{ $index = $i }}{{ end -}} {{- end -}} {{- if ge $index 0 -}} {{- $out = dict "name" $name "parts" $parts "index" $index "total" (len $parts) "overview" (site.GetPage (printf "/series/%s" (urlize $name))) -}} {{- end -}} {{- end -}} {{- return $out -}}
