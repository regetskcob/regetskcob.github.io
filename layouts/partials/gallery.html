{{/* Photo galleries, driven by data/gallery.yaml. Renders one titled block per entry under "galleries", in file order, so a new theme is added by appending a block to that file — no template change needed. Within a block, an item with "post" + "image" pulls the file out of that blog post's page bundle and links the tile to the post; an item with only "image" reads assets/gallery/. Missing files are skipped rather than failing the build, so a typo in the data file does not take the home page down. Images keep their own aspect ratio and flow into a masonry column layout: cropping everything to a uniform tile would cut roughly half the height off the portrait shots. "alt" describes the image for screen readers and is expected on every item. "caption" is separate and optional: it renders visibly under the tile, so it is only worth setting where there is something to add beyond the description. */}} {{- $blocks := slice -}} {{/* $ is the template root inside range, so bind items explicitly. */}} {{- range $g := site.Data.gallery.galleries -}} {{- $tiles := slice -}} {{- range $it := $g.items -}} {{- $img := "" -}} {{- with $it.post -}} {{- with site.GetPage (printf "/blog/%s" .) -}} {{- $img = .Resources.Get $it.image -}} {{- end -}} {{- else -}} {{- $img = resources.Get (printf "gallery/%s" $it.image) -}} {{- end -}} {{- with $img -}} {{- $tiles = $tiles | append (dict "img" . "alt" $it.alt "caption" $it.caption "post" $it.post) -}} {{- end -}} {{- end -}} {{- with $tiles -}} {{- $blocks = $blocks | append (dict "title" $g.title "description" $g.description "tiles" .) -}} {{- end -}} {{- end -}} {{- with $blocks }}
