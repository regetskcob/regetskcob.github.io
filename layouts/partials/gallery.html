{{/* Photo galleries on the home page, driven by data/gallery.yaml. Renders one titled block per entry under "galleries", in file order, so a new theme is added by appending a block to that file, no template change needed. Each block uses the shared photo grid (partials/photo-grid.html): two to four columns, breaking out of the text column, a random order on every visit and a lightbox on click. "size: large" on an item makes it span 2x2. Each block, title and description included, uses the wide gallery width; its title is an h2 like the series teaser's, the blocks sit on the same level. An item with "post" + "image" pulls the file out of that blog post's page bundle and links the tile to the post instead of the lightbox; an item with only "image" reads assets/gallery/. Missing files are skipped rather than failing the build, so a typo in the data file does not take the home page down. "alt" describes the image and is expected on every item. */}} {{- $blocks := slice -}} {{/* $ is the template root inside range, so bind items explicitly. */}} {{- range $g := site.Data.gallery.galleries -}} {{- $items := slice -}} {{- range $it := $g.items -}} {{- $img := "" -}} {{- $href := "" -}} {{- with $it.post -}} {{- with site.GetPage (printf "/blog/%s" .) -}} {{- $img = .Resources.Get $it.image -}} {{- $href = .RelPermalink -}} {{- end -}} {{- else -}} {{- $img = resources.Get (printf "gallery/%s" $it.image) -}} {{- end -}} {{- with $img -}} {{- $items = $items | append (dict "img" . "alt" $it.alt "size" $it.size "href" $href) -}} {{- end -}} {{- end -}} {{- with $items -}} {{- $blocks = $blocks | append (dict "title" $g.title "description" $g.description "items" .) -}} {{- end -}} {{- end -}} {{- with $blocks }}
