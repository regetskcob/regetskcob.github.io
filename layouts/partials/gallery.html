{{/* Photo gallery, driven by data/gallery.yaml. An entry with "post" + "image" pulls the file out of that blog post's page bundle and links the tile to the post; an entry with only "image" reads assets/gallery/ and renders an unlinked tile. Missing files are skipped rather than failing the build, so a typo in the data file does not take the home page down. Tiles are cropped to a uniform 3:2 so the grid stays even regardless of the source aspect ratios. */}} {{- $items := slice -}} {{/* $ is the template root inside range, so bind the item explicitly. */}} {{- range $it := site.Data.gallery.items -}} {{- $img := "" -}} {{- with $it.post -}} {{- with site.GetPage (printf "/blog/%s" .) -}} {{- $img = .Resources.Get $it.image -}} {{- end -}} {{- else -}} {{- $img = resources.Get (printf "gallery/%s" $it.image) -}} {{- end -}} {{- with $img -}} {{- $items = $items | append (dict "img" . "caption" $it.caption "post" $it.post) -}} {{- end -}} {{- end -}} {{- with $items }}

{{ with site.Params.galleryTitle }}{{ . }}{{ else }}Fotografie{{ end }}

{{- end }}
