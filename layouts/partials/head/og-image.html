{{/* Open Graph image, returned as an absolute URL string. The theme's opengraph.html calls this only as a fallback: Hugo's built-in _funcs/get-page-images runs first and wins whenever a page declares "images" in front matter or carries a resource named *cover* / *feature* / *thumbnail*. Everything else — the home page, sections, and posts whose images are named plainly — lands here. Order: the image named in front matter (cover / featured_image, carried over from the Ghost export), else the first image of the page bundle, else the site-wide fallback (ogImage in hugo.toml). All are cropped to 1200x630, the ratio Facebook, LinkedIn, Slack and WhatsApp expect. */}} {{- /* JPEG, not WebP: LinkedIn and some messengers skip WebP previews, and WhatsApp only shows images below about 300 KB. */ -}} {{- $spec := "1200x630 jpg q75 smart" -}} {{- $og := "" -}} {{/* Only raster images can be cropped; skip SVG and animated GIF. */}} {{- $raster := slice -}} {{- range .Resources.ByType "image" -}} {{- if not (in (slice "svg" "gif") .MediaType.SubType) -}} {{- $raster = $raster | append . -}} {{- end -}} {{- end -}} {{/* A series lists its photos in front matter; the first one represents it. */}} {{- with .Params.photos -}} {{- with resources.Get (printf "gallery/%s" (index . 0).image) -}} {{- with .Fill $spec -}} {{- $og = .Permalink -}} {{- end -}} {{- end -}} {{- end -}} {{/* An explicit choice in front matter wins over bundle order. */}} {{- if not $og -}}{{- with or .Params.cover .Params.featured_image -}} {{- $named := $.Resources.Get (strings.TrimPrefix "./" .) -}} {{- with $named -}} {{- if not (in (slice "svg" "gif") .MediaType.SubType) -}} {{- with .Fill $spec -}} {{- $og = .Permalink -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}{{- end -}} {{- if not $og -}} {{- with $raster -}} {{- with (index . 0).Fill $spec -}} {{- $og = .Permalink -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $og -}} {{- with site.Params.ogImage | default "images/og-image.jpg" | resources.Get -}} {{- with .Fill $spec -}} {{- $og = .Permalink -}} {{- end -}} {{- end -}} {{- end -}} {{- $og -}}
