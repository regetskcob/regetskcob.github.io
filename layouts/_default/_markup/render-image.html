{{/* Image render hook with a responsive resize pipeline. Overrides the theme's hook (themes/typo/layouts/_default/_markup/render-image.html), which serves originals untouched. Keeps the theme's DOM and class contract intact: figure > .img-container[--w/--h] > img[width][height], plus the img- classes derived from the "image.jpg#tag" suffix. Originals are resized to a srcset and converted to WebP. Non-raster formats (SVG, GIF) and remote images are passed through untouched. */}} {{- $u := urls.Parse .Destination -}} {{- $page := .PageInner -}} {{- $img := "" -}} {{/* Resolve the image against the page bundle, then the parent bundle, then assets. */}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $img = $page.Resources.Get $path -}} {{- if not $img -}} {{- with $page.Parent -}} {{- $img = .Resources.Get $path -}} {{- end -}} {{- end -}} {{- if not $img -}} {{- $img = resources.Get $path -}} {{- end -}} {{- end -}} {{/* "image.jpg#small#light" -> "img-small img-light" */}} {{- $classes := slice -}} {{- with $u.Fragment -}} {{- range split . "#" -}} {{- with . }}{{ $classes = $classes | append (printf "img-%s" .) }}{{ end -}} {{- end -}} {{- end -}} {{- $class := delimit $classes " " -}} {{- $widths := slice 480 800 1200 1600 -}} {{- $sizes := "(max-width: 780px) 100vw, 780px" -}} {{- if in $classes "img-full" -}} {{- $sizes = "100vw" -}} {{- end -}}
